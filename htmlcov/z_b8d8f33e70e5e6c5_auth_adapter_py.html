1"""The built-in adapter, for single-tenant projects that use Django's own users, groups and permissions. 

2 

3It is used when ``APPROVALS_TENANT_MODEL`` and ``APPROVALS_ADAPTER`` are not set: 

4 

5- every active user belongs to the one built-in tenant (``Tenant.default()``); 

6- roles are Django groups (``auth.Group``); 

7- permissions are ordinary Django permissions in the ``approvals`` app (``approvals.approve_<action_type>``, 

8 ``approvals.manage_approval_policies``), created after every ``migrate`` by ``sync_permissions`` and granted to 

9 users or groups as usual; 

10- notifications are emailed with ``django.core.mail`` to recipients that have an email address; 

11- audit records are written to the ``django_maker_checker.audit`` logger. 

12 

13Subclass it to change one of these, e.g. ``request_url`` for links in notification emails. 

14""" 

15

35class AuthApprovalsAdapter(ApprovalsAdapter): 

36 """Maker-checker on Django's ``auth`` app, with one tenant.""" 

37 

38 # --- tenancy and roles ---------------------------------------------------------------------- 

39 

40 def tenant_of(self, user): 

41 if not getattr(user, "is_authenticated", False) or not user.is_active: 

42 return None 

43 return Tenant.default() 

44 

45 def members(self, tenant, user_ids=None, role_ids=None): 

46 users = get_user_model().objects.filter(is_active=True) 

47 if user_ids is not None or role_ids is not None: 

48 users = users.filter(Q(pk__in=list(user_ids or ())) | Q(groups__in=list(role_ids or ()))) 

49 return users.distinct() 

50 

51 def role_ids_of(self, user): 

52 return set(user.groups.values_list("pk", flat=True)) 

53 

54 def roles(self, tenant): 

55 return Group.objects.all() 

56 

57 def role_member_count(self, role): 

58 return role.user_set.filter(is_active=True).count() 

59 

60 # --- permissions ---------------------------------------------------------------------------- 

61 

62 def has_permission(self, user, codename): 

63 return user.has_perm(f"{APP_LABEL}.{codename}") 

64 

65 def role_has_permission(self, role, codename): 

66 return role.permissions.filter(content_type__app_label=APP_LABEL, codename=codename).exists() 

67 

68 # --- side effects --------------------------------------------------------------------------- 

69 

70 def notify(self, notice): 

71 emails = [user.email for user in notice.recipients if getattr(user, "email", "")] 

72 if not emails: 

73 return 

74 body = f"{notice.text}\n\n{notice.link}" if notice.link else notice.text 

75 send_mail(notice.title, body, None, emails) 

76 

77 def audit(self, record): 

78 audit_logger.info( 

79 "%s %s %s", record.action, record.resource_type, record.resource_id, extra={"audit": asdict(record)} 

80 )

83def sync_permissions(using="default", **kwargs): 

84 """Create the ``approvals`` permissions the built-in adapter checks: one per registered action, plus the 

85 administrators' permission. Runs after every ``migrate``, so new actions get their permission automatically.""" 

86 content_type = ContentType.objects.db_manager(using).get_for_model(ApprovalPolicy) 

87 # Permission names are stored in the database, so they stay in English like Django's own. 

88 wanted = {MANAGE_POLICIES: "Manage maker-checker approval rules"} 

89 wanted.update({handler.approve_permission: handler.approve_permission_label for handler in all_handlers()}) 

90 for codename, name in wanted.items(): 

91 Permission.objects.db_manager(using).get_or_create( 

92 content_type=content_type, codename=codename, defaults={"name": name[:255]} 

93 )
