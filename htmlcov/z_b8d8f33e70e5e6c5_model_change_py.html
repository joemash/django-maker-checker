1"""``ModelChangeHandler``: a ready-made handler for "change these fields on one record". 

2 

3This is the Template Method pattern. The base class owns the whole flow (lock per record, snapshot of the current 

4values, optimistic check that nobody changed them since submission, apply, Current/Proposed review rows); a subclass 

5only declares the model and the fields that may change (and, if needed, how to find a tenant's records). 

6 

7.. code-block:: python 

8 

9 @register 

10 class CreditLimitChangeHandler(ModelChangeHandler): 

11 action_type = "credit_limit_change" 

12 label = "Credit Limit Change" 

13 category = "Accounts" 

14 model = Account 

15 fields = ("credit_limit",) 

16 tenant_field = "organisation" 

17 

18 outcome = CreditLimitChangeHandler.submit_change(account, {"credit_limit": "5000.00"}, maker=request.user, 

19 justification="Annual review", request=request) 

20""" 

21 

22from django.core.exceptions import ValidationError 

23from django.utils.translation import gettext as _ 

24 

25from django_maker_checker import services 

26from django_maker_checker.actions import Outcome, to_payload 

27from django_maker_checker.exceptions import ApprovalError 

28from django_maker_checker.registry import ActionHandler, ExecutionContext, diff_rows, get_handler 

29 

30 

31class ModelChangeHandler(ActionHandler): 

32 """Handler for "change these fields on one record". Subclasses set ``model``, ``fields`` and, if the foreign key 

33 to the tenant isn't called ``tenant``, ``tenant_field``. 

34 

35 Payload shape: ``{"pk": …, "label": …, "changes": {field: JSON value}}``. 

36 """ 

37 

38 model = None 

39 fields: tuple = () 

40 tenant_field = "tenant" # the model's foreign key to APPROVALS_TENANT_MODEL 

41 

42 # --- what a subclass may override --------------------------------------------------------------- 

43 

44 def queryset(self, tenant): 

45 """Records of ``model`` that belong to ``tenant``. Must be tenant-scoped.""" 

46 return self.model._default_manager.filter(**{self.tenant_field: tenant}) 

47 

48 def tenant_of(self, obj): 

49 """The tenant that owns ``obj``; that tenant's approval rules apply.""" 

50 return getattr(obj, self.tenant_field) 

51 

52 def describe(self, obj) -> str: 

53 return str(obj) 

54 

55 def apply(self, obj, changes, context: ExecutionContext): 

56 """Write the approved values. Override to call a service instead of ``save()``.""" 

57 for name, value in changes.items(): 

58 setattr(obj, name, value) 

59 obj.full_clean(exclude=[f.name for f in obj._meta.fields if f.name not in changes]) 

60 obj.save(update_fields=list(changes)) 

61 

62 # --- the fixed flow -------------------------------------------------------------------------------- 

63 

64 @classmethod 

65 def submit_change(cls, obj, changes: dict, *, maker, justification="", request=None): 

66 """Route a change to ``obj`` through maker-checker (applies at once when no policy applies).""" 

67 handler = get_handler(cls.action_type) 

68 unknown = set(changes) - set(handler.fields) 

69 if unknown: 

70 raise ValueError(f"{cls.__name__} cannot change: {', '.join(sorted(unknown))}") 

71 payload = {"pk": obj.pk, "label": handler.describe(obj), "changes": to_payload(changes)} 

72 approval_request, result = services.submit_or_execute( 

73 cls.action_type, 

74 tenant=handler.tenant_of(obj), 

75 maker=maker, 

76 payload=payload, 

77 justification=justification, 

78 request=request, 

79 ) 

80 return Outcome(approval_request, result) 

81 

82 def lock_key(self, payload): 

83 return f"{self.model._meta.label_lower}:{payload['pk']}" 

84 

85 def target_label(self, payload): 

86 return payload.get("label", "") 

87 

88 def _record(self, tenant, pk): 

89 # Always looked up through the tenant-scoped queryset, so a payload can't reach another tenant's record. 

90 return self.queryset(tenant).filter(pk=pk).first() 

91 

92 def _current(self, obj, names): 

93 # Current values in payload (JSON) form, so they compare equal to the stored snapshot. 

94 return to_payload({name: getattr(obj, name) for name in names}) 

95 

96 def snapshot_before(self, tenant, payload): 

97 obj = self._record(tenant, payload["pk"]) 

98 return self._current(obj, payload["changes"]) if obj else {} 

99 

100 def validate(self, approval_request): 

101 """Optimistic concurrency check: the fields being changed must still hold the values captured at submission. 

102 Otherwise approving would silently overwrite someone else's edit, so the request fails instead.""" 

103 payload = approval_request.payload 

104 obj = self._record(approval_request.tenant, payload["pk"]) 

105 if obj is None: 

106 raise ApprovalError(_("%(action)s: the record no longer exists.") % {"action": self.label}) 

107 # Only snapshotted requests can detect concurrent edits; direct execution has no snapshot. 

108 before = getattr(approval_request, "before_snapshot", None) 

109 if before and self._current(obj, payload["changes"]) != before: 

110 raise ApprovalError( 

111 _("%(action)s: the record was changed after this request was raised.") % {"action": self.label} 

112 ) 

113 

114 def execute(self, approval_request, context: ExecutionContext): 

115 """Re-validate (the direct, no-policy path skips the services' validate step), convert JSON values back to 

116 field types, then ``apply``. Model validation errors become ApprovalError so they show as a failure reason.""" 

117 payload = approval_request.payload 

118 self.validate(approval_request) 

119 obj = self._record(approval_request.tenant, payload["pk"]) 

120 changes = {name: obj._meta.get_field(name).to_python(value) for name, value in payload["changes"].items()} 

121 try: 

122 self.apply(obj, changes, context) 

123 except ValidationError as err: 

124 raise ApprovalError("; ".join(err.messages)) from err 

125 return {"pk": obj.pk, "changed": sorted(changes)} 

126 

127 def review_context(self, approval_request): 

128 """Current/Proposed rows labelled with the fields' verbose names.""" 

129 labels = {f.name: str(f.verbose_name).capitalize() for f in self.model._meta.fields} 

130 return {"rows": diff_rows(approval_request.before_snapshot or {}, approval_request.payload["changes"], labels)}
